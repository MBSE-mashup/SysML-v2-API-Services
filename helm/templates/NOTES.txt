1. Get the application URL by running these commands:
{{- if .Values.app.route.enabled }}

  export APP_HOST=$(kubectl get route {{ include "sysml-v2-api-services.fullname" . }} -o jsonpath='{.spec.host}')
  echo "https://$APP_HOST"

{{- else }}

  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "sysml-v2-api-services.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  echo "Visit http://127.0.0.1:9000 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 9000:{{ .Values.app.service.port }}

{{- end }}
